مخطط قاعدة البيانات المحلية (SQLite / Room Schema — 9 جداول)
local_db_schema.sql 
-- 1. جدول الدوائر العائلية (Circles Table)
CREATE TABLE IF NOT EXISTS circles (
    circle_id TEXT PRIMARY KEY NOT NULL,
    circle_name TEXT NOT NULL,
    circle_color TEXT DEFAULT '#6835F9',
    invite_code TEXT UNIQUE,
    invite_code_expires_at INTEGER,
    created_at INTEGER NOT NULL,
    updated_at INTEGER NOT NULL,
    is_premium INTEGER DEFAULT 0,
    admin_user_id TEXT NOT NULL
);

-- 2. جدول أعضاء الدائرة (Circle Members Table)
CREATE TABLE IF NOT EXISTS circle_members (
    member_id TEXT PRIMARY KEY NOT NULL,
    circle_id TEXT NOT NULL,
    user_id TEXT NOT NULL,
    first_name TEXT NOT NULL,
    role TEXT CHECK(role IN ('ADMIN', 'MEMBER', 'CHILD')) DEFAULT 'MEMBER',
    battery_level INTEGER DEFAULT 100,
    is_charging INTEGER DEFAULT 0,
    driving_safety_score INTEGER DEFAULT 100,
    last_seen_timestamp INTEGER NOT NULL,
    FOREIGN KEY (circle_id) REFERENCES circles(circle_id) ON DELETE CASCADE
);

-- 3. جدول الأماكن والجيوفنسينغ (Places & Geofences Table)
CREATE TABLE IF NOT EXISTS places (
    place_id TEXT PRIMARY KEY NOT NULL,
    circle_id TEXT NOT NULL,
    name TEXT NOT NULL,
    latitude REAL NOT NULL,
    longitude REAL NOT NULL,
    radius_meters REAL NOT NULL DEFAULT 100.0,
    notify_on_enter INTEGER DEFAULT 1,
    notify_on_exit INTEGER DEFAULT 1
);

-- 4. جدول الرحلات وسلوك القيادة (Driving Trips Table)
CREATE TABLE IF NOT EXISTS trips (
    trip_id TEXT PRIMARY KEY NOT NULL,
    user_id TEXT NOT NULL,
    start_time INTEGER NOT NULL,
    end_time INTEGER NOT NULL,
    total_distance_km REAL NOT NULL,
    max_speed_kmh REAL NOT NULL,
    avg_speed_kmh REAL NOT NULL,
    hard_brake_count INTEGER DEFAULT 0,
    rapid_accel_count INTEGER DEFAULT 0,
    trip_score INTEGER NOT NULL DEFAULT 100
);
